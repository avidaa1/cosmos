/* ============ OFFLINE COSMOS CHATBOT (phase 1, KB expands later) ============ */
var COSMOS_KB = {
  'sun': 'The Sun holds 99.8% of the Solar System\'s mass and has a core temperature of about 15 million C.',
  'mercury': 'Mercury is the smallest planet and has no moons.',
  'venus': 'Venus is the hottest planet due to its thick CO2 atmosphere trapping heat.',
  'earth': 'Earth is the only known planet confirmed to host life.',
  'mars': 'Mars hosts Olympus Mons, the largest known volcano in the Solar System.',
  'jupiter': 'Jupiter is the largest planet and has at least 95 known moons.',
  'saturn': 'Saturn\'s rings are made almost entirely of ice particles.',
  'uranus': 'Uranus rotates nearly on its side, likely from an ancient collision.',
  'neptune': 'Neptune has the strongest winds in the Solar System, up to 2,100 km/h.',
  'pluto': 'Pluto was reclassified as a dwarf planet in 2006.',
  'asteroid': 'The asteroid belt sits between Mars and Jupiter and contains millions of rocky bodies.',
  'black hole': 'The black hole here is an illustrative model. Real black hole shadows have been imaged by the Event Horizon Telescope (M87*, SgrA*).',
  'wormhole': 'Wormholes are theoretical -- permitted by general relativity but never observed.',
  'libra': 'Libra is a constellation with no first-magnitude stars; its brightest, Zubeneschamali, shines at magnitude 2.61. Full constellation mode ships in a future COSMOS update.'
};
function cosmosChatReply(q){
  q = q.toLowerCase();
  var keys = Object.keys(COSMOS_KB);
  for (var i=0;i<keys.length;i++){
    if (q.indexOf(keys[i]) !== -1) return COSMOS_KB[keys[i]];
  }
  return 'I don\'t have that yet in COSMOS -- try asking about a planet, the asteroid belt, the black hole, or the wormhole.';
}
window.toggleCosmosChat = function(){
  document.getElementById('cosmos-chat-panel').classList.toggle('show');
};
window.cosmosChatSend = function(){
  var input = document.getElementById('cosmos-chat-input');
  var q = input.value.trim();
  if (!q) return;
  var log = document.getElementById('cosmos-chat-log');
  var uDiv = document.createElement('div'); uDiv.className='msg-user'; uDiv.textContent = q;
  log.appendChild(uDiv);
  var bDiv = document.createElement('div'); bDiv.className='msg-bot'; bDiv.textContent = cosmosChatReply(q);
  log.appendChild(bDiv);
  log.scrollTop = log.scrollHeight;
  input.value = '';
};
document.getElementById('cosmos-chat-input').addEventListener('keydown', function(e){
  if (e.key === 'Enter') window.cosmosChatSend();
});
